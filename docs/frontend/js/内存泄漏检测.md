    <div class="steps">
      <h3>📋 使用步骤:</h3>
      <ol>
        <li><strong>打开 Chrome DevTools:</strong> 按 F12 或右键 → 检查</li>
        <li><strong>切换到 Memory 面板</strong></li>
        <li><strong>第一次快照:</strong> 选择 "Heap snapshot" → 点击 "Take snapshot" (拍摄初始快照)</li>
        <li><strong>触发内存泄漏:</strong> 点击下方的 "创建内存泄漏" 按钮多次 (建议 5-10 次)</li>
        <li><strong>第二次快照:</strong> 再次点击 "Take snapshot" (拍摄操作后的快照)</li>
        <li><strong>对比分析:</strong>
          <ul>
            <li>选择第二个快照</li>
            <li>在顶部下拉菜单中选择 "Comparison" (对比)</li>
            <li>在 "Comparison" 旁边选择 "Snapshot 1" 作为基准</li>
            <li>查看 "# Delta" 列,找到增长的对象</li>
            <li>搜索 "LeakyClass" 或 "detached" 查看泄漏的对象</li>

        </li>
        <li><strong>修复后验证:</strong> 点击 "清理泄漏" → 再拍快照 → 对比查看内存是否下降</li>

    <div class="container">
      <h2>场景 1: 全局变量引用导致的泄漏</h2>
      <p>每次点击都会创建大量对象并存储在全局数组中,无法被垃圾回收</p>
      <button onclick="createLeak()">🔴 创建内存泄漏</button>

    <div class="container">
      <h2>场景 4: DOM 分离节点泄漏</h2>
      <p>删除 DOM 节点但仍持有引用,导致分离的 DOM 无法回收</p>

    <script>
      // ============ 场景 1: 全局变量引用泄漏 ============
      let leakyArray = [];

      class LeakyClass {
        constructor(size) {
          this.data = new Array(size).fill('🔴 这是会造成内存泄漏的数据 ');
          this.timestamp = new Date().toLocaleTimeString();
        }
      }

      function createLeak() {
        // 每次创建 1000 个对象,每个对象包含大量数据
        for (let i = 0; i < 1000; i++) {
          leakyArray.push(new LeakyClass(100));
        }
        document.getElementById('status').textContent =
          `已创建 ${leakyArray.length} 个泄漏对象 (约 ${(leakyArray.length * 100 * 50 / 1024 / 1024).toFixed(2)} MB)`;
      }

      function clearLeak() {
        leakyArray = [];
        document.getElementById('status').textContent = '✅ 泄漏已清理!';
      }

      // ============ 场景 2: 事件监听器泄漏 ============
      let eventListeners = [];
      let listenerCount = 0;

      function createEventListenerLeak() {
        const bigData = new Array(10000).fill('大数据对象');

        const handler = function() {
          // 这个闭包会持有 bigData 的引用
          console.log('点击事件触发,持有数据:', bigData.length);
        };

        document.body.addEventListener('click', handler);
        eventListeners.push(handler);
        listenerCount++;

        document.getElementById('listenerStatus').textContent =
          `已添加 ${listenerCount} 个未清理的事件监听器`;
      }

      function clearEventListeners() {
        eventListeners.forEach(handler => {
          document.body.removeEventListener('click', handler);
        });
        eventListeners = [];
        document.getElementById('listenerStatus').textContent =
          `✅ 已清理 ${listenerCount} 个事件监听器`;
        listenerCount = 0;
      }

      // ============ 场景 3: 定时器泄漏 ============
      let timers = [];

      function createTimerLeak() {
        const bigData = new Array(50000).fill('定时器持有的大数据');

        const timerId = setInterval(() => {
          // 这个闭包会持有 bigData 的引用
          console.log('定时器执行,数据大小:', bigData.length);
        }, 1000);

        timers.push(timerId);

        document.getElementById('timerStatus').textContent =
          `已创建 ${timers.length} 个未清理的定时器`;
      }

      function clearTimers() {
        const count = timers.length;
        timers.forEach(timerId => {
          clearInterval(timerId);
        });
        timers = [];
        document.getElementById('timerStatus').textContent =
          `✅ 已清理 ${count} 个定时器`;
      }

      // ============ 场景 4: 分离 DOM 节点泄漏 ============
      let detachedNodes = [];

      function createDetachedDOMLeak() {
        const container = document.getElementById('domContainer');

        // 创建大量 DOM 节点
        for (let i = 0; i < 100; i++) {
          const div = document.createElement('div');
          div.innerHTML = `<p>节点 ${i}</p>` + '<span>子节点</span>'.repeat(50);
          container.appendChild(div);

          // 保存引用
          detachedNodes.push(div);
        }

        // 从 DOM 树中移除,但仍持有引用
        container.innerHTML = '';

        document.getElementById('domStatus').textContent =
          `已创建 ${detachedNodes.length} 个分离的 DOM 节点(已从 DOM 树移除但仍持有引用)`;
      }

      function clearDetachedDOM() {
        const count = detachedNodes.length;
        detachedNodes = [];
        document.getElementById('domContainer').innerHTML = '';
        document.getElementById('domStatus').textContent =
          `✅ 已清理 ${count} 个分离 DOM 节点的引用`;
      }

      // 初始化提示
      console.log('%c内存泄漏检测演示已加载!', 'color: green; font-size: 16px; font-weight: bold;');
      console.log('请按照页面上的步骤使用 Chrome DevTools Memory 面板进行检测');
    </script>
